一、策略思路评价
1.1 策略定位
名称：基于小市值，近期涨跌幅和成交额缩波因子-filter-ROE,归母净利润增速的择时动态模式切换
核心逻辑：小市值 + 近期涨跌幅+ 成交额缩波，择时：动态模式切换（参考走天涯大佬的择时策略）
持仓周期：5D
选股数量：10

回测时间：2009-01-01~2026-06-05

最优因子参数，包括择时参数 都是通过beam search+lsh+细网格搜索出来的。

1.2 config

strategy = {
    'name': '市值-近期涨跌-成交缩波-5D-num10-beam策略NO.2-择时-动态模式切换-NO.1',  # 策略名
    'hold_period': '5D',  # 持仓周期，W 代表周，M 代表月，还支持日频：3D、5D、10D
    'select_num': 10,  # 选股数量，可以是整数，也可以是小数，比如 0.1 表示选取 10% 的股票
    "factor_list": [  # 选股因子列表
        ('成交额缩波因子', True, (12, 60),0.2),
        ('近期涨跌幅', True, 5, 0.3),   
        ('市值', True, None, 1.0), 
    ],
    "filter_list": [
        ('ROE', '单季', 'pct:<0.7', False),
        ('归母净利润同比增速', 60, 'pct:<0.7', False),
   
        ('收盘价', '', 'val:>=3'),  # 股价不低于3元，避免低价垃圾股
        ('收盘价', '', 'val:<=40'),  # 股价不高于40元，控制价格区间
        ('剔除ST', '', 'val:==1'),  # 剔除ST、*ST等风险警示股票

    ]  # 过滤因子列表
}

days_listed = 250  # 上市至今交易天数

excluded_boards = []  # 过滤板块，默认不过滤

equity_timing = {"name": "动态模式切换", "params": ['sh000300', 24, 25, 10,20,15,2.0]}
