（2）Donchian通道位置_20日

import pandas as pd

fin_cols = []

def add_factor(df: pd.DataFrame, param, fin_data=None, **kwargs) -> (pd.DataFrame, dict):

    """
    因子说明：Donchian通道位置_20日
    计算逻辑：
    1. 计算过去20日最高价与最低价，构建Donchian通道上下轨
    2. 计算通道中轨 = (上轨+下轨)/2
    3. 计算当前收盘价在通道中的相对位置 = (收盘价-中轨)/(上轨-下轨+1e-8)
  
    物理意义：
    衡量当前价格在20日价格区间内的相对强弱位置。正值表示价格位于中轨上方，负值表示位于中轨下方。在中国A股市场，该因子反映了短期趋势强度与超买超卖状态，结合反身性理论，当价格持续位于通道上轨时可能引发追涨效应，位于下轨时可能触发恐慌性抛售，但极端位置也往往预示着反转机会。
    """
    n = int(param)  # 滚动周期数
    col_name = kwargs['col_name']
    hh, ll = df['最高价_复权'].rolling(n).max(), df['最低价_复权'].rolling(n).min()
    mid = (hh + ll) / 2
    factor_col = (df['收盘价_复权'] - mid) / (hh - ll + 1e-8)

    factor_df = pd.DataFrame({col_name: factor_col}, index=df.index)

    agg_dict = {col_name: 'last'}
    return factor_df, agg_dict

import pandas as pd

fin_cols = []

def add_factor(df: pd.DataFrame, param, fin_data=None, **kwargs) -> (pd.DataFrame, dict):

    """
    因子说明：枢轴点强度_20日
    计算逻辑：
    1. 计算20日最高价、最低价、收盘价复权均值的1/3作为枢轴点（Pivot Point）
    2. 计算当前收盘价相对于枢轴点的偏离度：(收盘价 - 枢轴点) / 枢轴点
  
    物理意义：
    基于经典枢轴点理论，衡量股价在20日周期内相对于量价均衡中枢的偏离强度。在中国A股市场，该因子捕捉了短期价格反身性效应：当股价显著高于枢轴点时，反映市场情绪过热和趋势惯性，但同时也暗示均值回归压力增大；反之，低于枢轴点则反映恐慌性超卖。该因子在震荡市中具有较好的反转预测能力，在趋势市中则体现动量延续特征。
    """
    n = int(param)  # 滚动周期数
    col_name = kwargs['col_name']
    pp = (df['最高价_复权'] + df['最低价_复权'] + df['收盘价_复权']).rolling(n).mean() / 3
    factor_col = (df['收盘价_复权'] - pp) / (pp + 1e-8)

    factor_df = pd.DataFrame({col_name: factor_col}, index=df.index)

    agg_dict = {col_name: 'last'}
    return factor_df, agg_dict

（4）穿刺形态信号

import pandas as pd

fin_cols = []

def add_factor(df: pd.DataFrame, param, fin_data=None, **kwargs) -> (pd.DataFrame, dict):

    """
    因子说明：穿刺形态信号
    计算逻辑：
    1. 识别前一日为阴线（收盘<开盘）且当日为阳线（收盘>开盘），同时当日开盘价低于前日收盘价，且当日收盘价高于前日K线实体中点，定义为穿刺形态（看涨信号），赋值为+1。
    2. 识别前一日为阳线（收盘>开盘）且当日为阴线（收盘<开盘），同时当日开盘价高于前日收盘价，且当日收盘价低于前日K线实体中点，定义为乌云盖顶（看跌信号），赋值为-1。
    3. 其他情况赋值为0，最终因子值为穿刺信号减去乌云盖顶信号。
  
    物理意义：
    基于A股市场量价博弈与反身性理论，穿刺形态反映多头在空头主导后强势反攻，突破前日实体中点，预示短期趋势反转向上；乌云盖顶则相反，体现空头在高位压制后反噬，暗示趋势转弱。该因子捕捉市场情绪转折点，在震荡市中具有显著预测能力。
    """
    col_name = kwargs['col_name']
    prev_red = df['收盘价_复权'].shift(1) > df['开盘价_复权'].shift(1)
    prev_green = df['收盘价_复权'].shift(1) < df['开盘价_复权'].shift(1)
    today_green = df['收盘价_复权'] < df['开盘价_复权']
    today_red = df['收盘价_复权'] > df['开盘价_复权']
    mid = (df['开盘价_复权'].shift(1) + df['收盘价_复权'].shift(1)) / 2
    piercing = prev_green & today_red & (df['开盘价_复权'] < df['收盘价_复权'].shift(1)) & (df['收盘价_复权'] > mid)
    dark_cloud = prev_red & today_green & (df['开盘价_复权'] > df['收盘价_复权'].shift(1)) & (df['收盘价_复权'] < mid)
    factor_col = piercing.astype(float) - dark_cloud.astype(float)

    factor_df = pd.DataFrame({col_name: factor_col}, index=df.index)

    agg_dict = {col_name: 'last'}
    return factor_df, agg_dict

    因子说明：收盘价
    计算逻辑：
    1. 直接提取原始数据中的'收盘价'字段作为因子值。
    2. 因子聚合方式为取最后一个交易日的数据（'last'）。
  
    物理意义：
    收盘价是A股市场多空双方在日内博弈的最终均衡价格，反映了市场参与者对股票价值的瞬时共识。在中国A股市场，收盘价常受尾盘资金博弈、机构调仓及散户情绪影响，具有反身性特征——价格变动会进一步强化或扭转市场预期，从而影响次日开盘走势。该因子是技术分析和量化模型的基础输入，常用于计算收益率、波动率及各类衍生因子。
    """

    col_name = kwargs['col_name']

    # 我们这里的市值因子使用总市值的数值
    factor_col = df['收盘价']

    factor_df = pd.DataFrame({col_name: factor_col}, index=df.index)

    agg_dict = {col_name: 'last'}

    # 返回新计算的因子列以及因子聚合方式
    return factor_df, agg_dict

import pandas as pd

# 财务因子列：此列表用于存储财务因子相关的列名称
fin_cols = []

def add_factor(df: pd.DataFrame, param, fin_data=None, **kwargs) -> (pd.DataFrame, dict):
    """
    因子说明：换手率
    计算逻辑：
    1. 计算每日的换手率 = 当日成交额 / 当日流通市值。
    2. 对每日换手率进行 param（如 5日、20日）周期的移动平均（rolling mean）。
  
    物理意义：
    衡量股票在特定周期内的交投活跃度。
    高换手率通常代表资金关注度高、流动性好，但也可能伴随高波动或情绪过热；
    低换手率则代表股票交投清淡。在不同宏观状态下，换手率因子的有效性方向会发生反转。
    """

    col_name = kwargs['col_name']
    # 我们这里的市值因子使用总市值的数值
    turnover_rate = df['成交额'] / df['流通市值']
    factor_col = turnover_rate.rolling(param).mean()

    factor_df = pd.DataFrame({col_name: factor_col}, index=df.index)

    agg_dict = {col_name: 'last'}

    # 返回新计算的因子列以及因子聚合方式
    return factor_df, agg_dict

    Alpha4因子定义：-1 * 最低价的时间序列排名（默认窗口9天）
    逻辑说明：通过计算一段时间内最低价在该区间的排名（1为最低，N为最高），再取相反数，反映价格低位特征
    """

    # ======================== 参数处理 ===========================
    # 从kwargs中提取因子列的名称
    col_name = kwargs['col_name']
    param = int(param[0])  # 时间序列排名窗口，可通过param调整
    # 设置默认参数（Alpha4通常使用9天排名窗口）
    # if param is None:
    #     param = 9  # 时间序列排名窗口，可通过param调整

    # ======================== 计算因子 ===========================
    # 提取最低价数据
    low_price = df['最低价']
  
    # 计算时间序列排名：在param天窗口内，当前最低价的排名（1表示窗口内最低，param表示最高）
    # min_periods=1确保窗口不足时仍能计算（窗口内有1个数据即可）
    ts_rank = low_price.rolling(window=param, min_periods=1).rank(method='min')  # method='min'确保排名稳定
  
    # Alpha4公式：-1 * 时间序列排名
    df[col_name] = -1 * ts_rank

    # ======================== 聚合方式 ===========================
    # 因子聚合方式：周期转换时保留最新值（价量类因子通用处理）
    agg_rules = {
        col_name: 'last'
    }

    return df[[col_name]], agg_rules

import pandas as pd
import numpy as np

# 财务因子列：此列表用于存储财务因子相关的列名称
fin_cols = []  # 财务因子列，配置后系统会自动加载对应的财务数据


def add_factor(df: pd.DataFrame, param=None, **kwargs) -> (pd.DataFrame, dict):
    """
    计算并将Alpha84因子添加到股票行情数据中，并返回包含计算因子的DataFrame及其聚合方式。
  
    Alpha84因子定义：VWAP与滚动窗口内最大值的差的时间序列排名，进行符号保持幂运算（幂为收盘价5天差分）
    逻辑说明：通过价格波动方向（收盘价差分）调节VWAP低位特征的权重，强化趋势与价格位置的联动
    """

    # ======================== 参数处理 ===========================
    # 从kwargs中提取因子列的名称
    col_name = kwargs['col_name']
    param1 = int(param[0])   # VWAP取最大值的窗口（原Alpha84为15天）
    param2 = int(param[1])  # 收盘价差分窗口（原Alpha84为5天）
    # 参数为字典，包含vwap_max_window（VWAP最大值窗口）和close_delta_window（收盘价差分窗口）
    # if param is None:
    #     param = {
    #         'vwap_max_window': 15,  # VWAP取最大值的窗口（原Alpha84为15天）
    #         'close_delta_window': 5  # 收盘价差分窗口（原Alpha84为5天）
    #     }

    # ======================== 计算因子 ===========================
    # 1. 计算VWAP（成交量加权平均价）
    # VWAP公式：sum(成交量*(最高价+最低价+收盘价)/3) / sum(成交量)
    typical_price = (df['最高价'] + df['最低价'] + df['收盘价']) / 3
    vwap = (df['成交量'] * typical_price).cumsum() / df['成交量'].cumsum()
    # 填充VWAP计算初期的NaN（首日VWAP等于当日典型价格）
    vwap = vwap.fillna(typical_price)

    # 2. 计算VWAP与滚动窗口内最大值的差（vwap - ts_max(vwap, window)）
    vwap_max = vwap.rolling(window=param1, min_periods=1).max()
    vwap_diff = vwap - vwap_max  # 结果为非正数（VWAP <= 窗口内最大值）

    # 3. 计算上述差值的时间序列排名（窗口内值越小，排名越靠前）
    ts_rank = vwap_diff.rolling(window=param1, min_periods=1).rank(method='min')

    # 4. 计算收盘价的N天差分（close - delay(close, N)）
    close_delta = df['收盘价'] - df['收盘价'].shift(param2)

    # 5. Alpha84公式：对时间序列排名进行符号保持幂运算（底数^指数）
    # 注：使用np.sign保留符号，避免负数开偶次幂出现复数
    df[col_name] = np.sign(ts_rank) * np.abs(ts_rank) ** close_delta

    # ======================== 聚合方式 ===========================
    # 因子聚合方式：周期转换时保留最新值（价量衍生因子通用处理）
    agg_rules = {
        col_name: 'last'
    }

    return df[[col_name]], agg_rules

# -*- coding: utf-8 -*-
"""
唐奇安Donchian通道位置指数择时【均线分区双参数版，蓝本来自你修改后的单参数代码】
参数传入顺序：n,long_thresh,short_thresh,junxian,n1,long_thresh1,short_thresh1
分区规则：
1、指数收盘 > MA_junxian(均线上)：通道周期n，阈值long_thresh / short_thresh
2、指数收盘 ≤ MA_junxian(均线下)：通道周期n1，阈值long_thresh1 / short_thresh1
信号规则（完全保留你修改后的特殊开仓逻辑）：
【开仓做多1】二选一满足即开仓
   ① 通道位置 上穿 long_thresh（今日>阈值，昨日≤阈值）
   ② 通道位置 上穿 short_thresh（今日>阈值，昨日≤阈值）
【空仓0两种场景】任意满足即空仓
   ① 通道位置 下穿 long_thresh（强势结束止盈离场）
   ② 通道位置 下穿 short_thresh（弱势破位止损）
3、无交叉触发延续昨日信号，初始默认空仓0
计算标的：880823微盘股指数（最高/最低/收盘）
因子：donch_pos = (收盘-通道中轨)/(通道上下轨差值)
"""
import pandas as pd
import numpy as np
from datetime import datetime as dt

def equity_signal(equity_df: pd.DataFrame, *args) -> pd.Series:
    # 读取外部7个参数
    n = int(args[0])
    long_thresh = float(args[1])
    short_thresh = float(args[2])
    junxian = int(args[3])
    n1 = int(args[4])
    long_thresh1 = float(args[5])
    short_thresh1 = float(args[6])

    # 1、读取微盘股指数880823
    index_df = pd.read_excel(
        r"C:\Users\Hasee\Desktop\880823.xls",
        sheet_name=0,
        engine="xlrd"
    )
    index_df.columns = index_df.columns.str.strip()
    index_df["时间"] = pd.to_datetime(index_df["时间"])
    index_df = index_df.rename(columns={"时间": "交易日期"})
    index_df = index_df.sort_values("交易日期").reset_index(drop=True)
    # 日期字符串索引，对齐资金曲线
    index_df["交易日期_str"] = index_df["交易日期"].dt.strftime("%Y-%m-%d")
    index_df = index_df.set_index("交易日期_str").sort_index()

    # 计算分界均线
    index_df["MA_X"] = index_df["收盘"].rolling(window=junxian, min_periods=1).mean()
    close = index_df["收盘"]
    ma_x = index_df["MA_X"]
    is_up = close > ma_x        # 均线上标记
    is_down = close <= ma_x     # 均线下标记

    # ========== 线上区间：周期n 全套指标 ==========
    hh_up = index_df["最高"].rolling(window=n, min_periods=1).max()
    ll_up = index_df["最低"].rolling(window=n, min_periods=1).min()
    mid_up = (hh_up + ll_up) / 2
    pos_up = (close - mid_up) / (hh_up - ll_up + 1e-8)

    # ========== 线下区间：周期n1 全套指标 ==========
    hh_down = index_df["最高"].rolling(window=n1, min_periods=1).max()
    ll_down = index_df["最低"].rolling(window=n1, min_periods=1).min()
    mid_down = (hh_down + ll_down) / 2
    pos_down = (close - mid_down) / (hh_down - ll_down + 1e-8)

    # ===================== 分区条件（完全复刻你原版买卖逻辑） =====================
    # -------------------- 线上信号 --------------------
    # 线上开仓：上穿强势阈值 OR 上穿弱势阈值
    buy_up = is_up & (
        ((pos_up > long_thresh) & (pos_up.shift(1) <= long_thresh))
        | ((pos_up > short_thresh) & (pos_up.shift(1) <= short_thresh))
    )
    # 线上空仓：下穿强势止盈 / 下穿弱势止损
    sell_up_profit = is_up & (pos_up < long_thresh) & (pos_up.shift(1) >= long_thresh)
    sell_up_stop = is_up & (pos_up < short_thresh) & (pos_up.shift(1) >= short_thresh)
    sell_up = sell_up_profit | sell_up_stop

    # -------------------- 线下信号 --------------------
    buy_down = is_down & (
        ((pos_down > long_thresh1) & (pos_down.shift(1) <= long_thresh1))
        | ((pos_down > short_thresh1) & (pos_down.shift(1) <= short_thresh1))
    )
    sell_down_profit = is_down & (pos_down < long_thresh1) & (pos_down.shift(1) >= long_thresh1)
    sell_down_stop = is_down & (pos_down < short_thresh1) & (pos_down.shift(1) >= short_thresh1)
    sell_down = sell_down_profit | sell_down_stop

    # 合并全市场开仓、空仓
    all_buy = buy_up | buy_down
    all_sell = sell_up | sell_down

    # 4、生成原始指数信号，空仓优先级高于开仓
    sig_raw = pd.Series(np.nan, index=index_df.index)
    sig_raw.loc[all_sell] = 0.0
    sig_raw.loc[all_buy] = 1.0
    # 无触发延续前值，初始默认空仓0
    sig_raw = sig_raw.ffill().fillna(0.0)

    # 5、信号对齐资金曲线日期
    result = pd.Series(0.0, index=equity_df.index)
    sig_map = sig_raw.to_dict()
    equity_df["交易日期"] = pd.to_datetime(equity_df["交易日期"], errors="coerce")

    for idx, row in equity_df.iterrows():
        dt_str = row["交易日期"].strftime("%Y-%m-%d")
        result.at[idx] = sig_map.get(dt_str, 0.0)

    # 6、控制台打印明细（增加均线、区间标记，方便核对分区）
    print("=" * 145)
    print(f"{'日期':<12}{'收盘':<8}{'分界MA':<10}{'区间':<6}{'线上通道位置':<12}{'线下通道位置':<12}{'信号':<6}")
    print("-" * 145)
    for _, row_eq in equity_df.iterrows():
        dt_str = row_eq["交易日期"].strftime("%Y-%m-%d")
        if dt_str in index_df.index:
            idx_row = index_df.loc[dt_str]
            close_val = round(idx_row["收盘"], 2)
            ma_val = round(ma_x.loc[dt_str], 2)
            zone = "线上" if is_up.loc[dt_str] else "线下"
            pos_u = round(pos_up.loc[dt_str], 6)
            pos_d = round(pos_down.loc[dt_str], 6)
        else:
            close_val = ma_val = pos_u = pos_d = "NaN"
            zone = "NaN"
        sig_val = int(result.loc[row_eq.name])
        print(f"{dt_str:<12}{close_val:<8}{ma_val:<10}{zone:<6}{pos_u:<12}{pos_d:<12}{sig_val:<6}")
    print("=" * 145)

    # 统计输出
    print(f"=== Donchian均线分区择时参数汇总 ===")
    print(f"分界均线周期：MA{junxian}")
    print(f"【均线上方】通道周期n={n}，强势阈值={long_thresh}，弱势阈值={short_thresh}")
    print(f"【均线下方】通道周期n1={n1}，强势阈值={long_thresh1}，弱势阈值={short_thresh1}")
    print(f"开仓规则：上穿强势阈值 OR 上穿弱势阈值 任意一个触发即开仓")
    print(f"空仓规则：下穿强势止盈 / 下穿弱势止损 任意一个触发即空仓")
    print(f"总交易日：{len(result)} | 做多(1)天数：{(result==1).sum()} | 空仓(0)天数：{(result==0).sum()}")
    return result
