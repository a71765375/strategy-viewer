def dict_itertools(dict_):
    # 小组框架特定代码，针对再择时做的优化
    dict_ = deepcopy(dict_)
    if "re_timing" in dict_:
        dict_.pop("re_timing")
    keys = list(dict_.keys())
    values = list(dict_.values())
    return [dict(zip(keys, combo)) for combo in itertools.product(*values)]
