一、策略思路

本策略做"小市值超跌反弹"的策略——回归动量，买的是离60日高点最远、跌得最惨的小盘股。利用A股价值回归的原则进行抄底。两个因子（小市值+超跌偏离回归），每3天换5只，2009年至今净值12,289倍，年化135.34%。

二、交易逻辑
选股因子（等权复合排名）

因子1：总市值 × 成交额10日标准差（升序）

取总市值与成交额10日标准差的乘积，值越小排名越靠前。

物理直觉：小市值且成交额波动小的股票，往往是被市场忽视的"冷门股"。A股题材炒作的发动机永远是小盘股——信息不对称程度高、机构覆盖少、筹码结构轻，一旦有催化剂，弹性远大于大市值股票。成交额标准差低，说明没有资金提前埋伏，爆发力更强。

因子2：动量因子（降序 → 实际升序）

原设计为20日动量（降序，追涨），但实证发现升序效果碾压降序。即选出最近20日跌得最多的小盘股。

物理直觉：这个因子原设计是捕捉突破追涨，但实证发现小盘股叠加超跌（跌得越惨越好），产生极强的均值回归弹性——跌透了的小盘股一旦反弹，空间巨大。这本质是"小市值 + 超跌反弹"，不是追涨突破。

过滤规则（三层预筛）
过滤	规则	目的
光底阴线	排除收盘价≈最低价的股票	去掉弱势股（无下影线=无支撑）
日均成交额	20日均成交额 ≥ 500万	保证流动性，排除僵尸股
市值门槛	总市值 ≥ 9亿	排除壳资源/退市边缘

板块与交易限制

排除：科创板(sh688)、北交所(bj)
初始资金100万，佣金万0.85，印花税千0.5

调仓节奏

每 3个交易日 调仓一次，每次持有 5只 股票

def load_data_from_folder():
    """从 stock-trading-data 文件夹加载所有CSV文件"""
    file_path = os.path.abspath(os.path.dirname(__file__)) + '/stock-trading-data/'
  
    if not os.path.exists(file_path):
        print(f"错误：文件夹不存在 {file_path}")
        return None
  
    file_list = [f for f in os.listdir(file_path) if f.endswith('.csv')]
  
    if len(file_list) == 0:
        print(f"错误：文件夹 {file_path} 中没有CSV文件")
        return None
  
    print(f"找到 {len(file_list)} 个CSV文件")
    print(f"回测区间：{START_DATE} ~ {END_DATE}")
  
    df_list = []
    for i, file in enumerate(file_list):
        if i % 500 == 0:
            print(f"正在读取第 {i+1}/{len(file_list)} 个文件...")
  
        temp_df = pd.read_csv(
            os.path.join(file_path, file),
            encoding='gbk',
            skiprows=1,
            low_memory=False
        )
  
        if '交易日期' in temp_df.columns:
            temp_df['交易日期'] = pd.to_datetime(temp_df['交易日期'])
            temp_df = temp_df[(temp_df['交易日期'] >= START_DATE) & 
                             (temp_df['交易日期'] <= END_DATE)]
            if len(temp_df) > 0:
                df_list.append(temp_df)
        else:
            print(f"警告：文件 {file} 中没有'交易日期'列，跳过")
  
    if len(df_list) == 0:
        print("错误：在回测区间内没有找到任何数据")
        return None
  
    df = pd.concat(df_list, ignore_index=True)
    print(f"总数据量: {len(df)} 行")
    print(f"实际日期范围: {df['交易日期'].min()} ~ {df['交易日期'].max()}")
  
    # 确保数值类型
    for col in ['开盘价', '最高价', '最低价', '收盘价', '前收盘价', '成交量', '成交额', '流通市值', '总市值']:
        if col in df.columns:
            df[col] = pd.to_numeric(df[col], errors='coerce')
  
    # 按股票代码和日期排序
    df = df.sort_values(['股票代码', '交易日期']).reset_index(drop=True)
  
    stock_count = df['股票代码'].nunique()
    print(f"股票数量: {stock_count}")
  
    # 若数据不含复权收盘价，则直接使用"收盘价"列（假设已复权）
    if '收盘价_复权' not in df.columns:
        df['收盘价_复权'] = df['收盘价']
  
    return df

def load_index_data(df):
    """从股票数据中提取沪深300和全A指数"""
    print("正在提取指数数据...")
  
    # 沪深300指数代码（通常为000300）
    hs300_data = df[df['股票代码'] == '000300'].copy()
  
    # 全A指数（使用全部股票的等权或流通市值加权平均）
    # 这里我们使用所有股票的平均价格作为全A指数代理
    all_a_data = df.groupby('交易日期').agg({
        '收盘价_复权': 'mean'
    }).reset_index()
    all_a_data['股票代码'] = 'ALL_A'
    all_a_data['股票名称'] = '全A指数'
  
    # 如果没有沪深300数据，尝试使用其他代码
    if len(hs300_data) == 0:
        # 尝试其他可能的沪深300代码
        possible_codes = ['sh000300', '000300.SH', 'hs300']
        for code in possible_codes:
            hs300_data = df[df['股票代码'] == code].copy()
            if len(hs300_data) > 0:
                break
  
    # 如果仍然没有沪深300数据，使用全A数据代替
    if len(hs300_data) == 0:
        print("警告：未找到沪深300指数数据，使用全A指数代替")
        hs300_data = all_a_data.copy()
        hs300_data['股票代码'] = 'HS300'
        hs300_data['股票名称'] = '沪深300(代理)'
  
    # 按日期排序
    hs300_data = hs300_data.sort_values('交易日期').reset_index(drop=True)
    all_a_data = all_a_data.sort_values('交易日期').reset_index(drop=True)
  
    print(f"沪深300数据量: {len(hs300_data)} 行")
    print(f"全A指数数据量: {len(all_a_data)} 行")
  
    return hs300_data, all_a_data

# ==================== 筛选股票函数 ====================
def filter_stocks(df):
    """
    筛除ST、北交所、科创板股票
    """
    print("正在筛除ST、北交所、科创板股票...")
    original_len = len(df)
  
    # 1. 筛除ST股票（股票名称包含ST或*ST）
    if '股票名称' in df.columns:
        # 创建ST标记
        df['is_st'] = df['股票名称'].str.contains('ST|\\*ST', na=False)
        before_filter = len(df)
        df = df[~df['is_st']]
        print(f"筛除ST股票: {before_filter - len(df)} 行被过滤，剩余 {len(df)} 行")
    else:
        print("警告：未找到'股票名称'列，无法筛除ST股票")
  
    # 2. 筛除北交所股票（代码以bj开头）
    if '股票代码' in df.columns:
        # 将股票代码转为字符串
        df['stock_code_str'] = df['股票代码'].astype(str)
        # 北交所代码以bj开头
        df['is_bj'] = df['stock_code_str'].str.startswith('bj')
        before_filter = len(df)
        df = df[~df['is_bj']]
        print(f"筛除北交所股票: {before_filter - len(df)} 行被过滤，剩余 {len(df)} 行")
  
    # 3. 筛除科创板股票（代码以sh688开头）
    if '股票代码' in df.columns:
        df['is_kcb'] = df['stock_code_str'].str.startswith('sh688')
        before_filter = len(df)
        df = df[~df['is_kcb']]
        print(f"筛除科创板股票: {before_filter - len(df)} 行被过滤，剩余 {len(df)} 行")
  
    # 删除临时列
    for col in ['is_st', 'is_bj', 'is_kcb', 'stock_code_str']:
        if col in df.columns:
            df = df.drop(columns=[col])
  
    print(f"共筛除 {original_len - len(df)} 行，剩余 {len(df)} 行")
    return df

# ==================== 因子计算函数 ====================
def calculate_factors(df, momentum_type='ret_20d'):
    """
    计算选股因子：
    因子1：总市值 * 成交额10日标准差（值越小越好）
    因子2：动量因子（值越大越好）
    """
    print("正在计算因子...")
    print(f"动量因子类型: {momentum_type}")
  
    df = df.sort_values(['股票代码', '交易日期'])
  
    # ---------- 因子1：总市值 * 成交额10日标准差 ----------
    df['amount_std_10'] = df.groupby('股票代码')['成交额'].rolling(10, min_periods=5).std().reset_index(level=0, drop=True)
  
    if '总市值' in df.columns:
        df['factor1_raw'] = df['总市值'] * df['amount_std_10']
    elif '流通市值' in df.columns:
        print("警告：未找到'总市值'列，使用'流通市值'替代")
        df['factor1_raw'] = df['流通市值'] * df['amount_std_10']
    else:
        raise KeyError("数据中既无'总市值'也无'流通市值'列，无法计算因子1")
  
    # ---------- 因子2：反转动量因子 ----------
    # 根据 momentum_type 计算不同周期的收益率
    if momentum_type == 'ret_20d':
        df['factor2_raw'] = df.groupby('股票代码')['收盘价_复权'].pct_change(periods=20) * 100
        print("使用动量因子：20日收益率")
    elif momentum_type == 'ret_60d':
        df['factor2_raw'] = df.groupby('股票代码')['收盘价_复权'].pct_change(periods=60) * 100
        print("使用动量因子：60日收益率")
    elif momentum_type == 'ret_120d':
        df['factor2_raw'] = df.groupby('股票代码')['收盘价_复权'].pct_change(periods=120) * 100
        print("使用动量因子：120日收益率")
    elif momentum_type == 'ret_5d_20d':
        df['ret_5d'] = df.groupby('股票代码')['收盘价_复权'].pct_change(periods=5) * 100
        df['ret_20d'] = df.groupby('股票代码')['收盘价_复权'].pct_change(periods=20) * 100
        df['factor2_raw'] = df['ret_5d'] / (df['ret_20d'].abs() + 1)   # 相对强弱
        print("使用动量因子：5日/20日收益率比值")
    elif momentum_type == 'ret_20d_60d':
        df['ret_20d'] = df.groupby('股票代码')['收盘价_复权'].pct_change(periods=20) * 100
        df['ret_60d'] = df.groupby('股票代码')['收盘价_复权'].pct_change(periods=60) * 100
        df['factor2_raw'] = df['ret_20d'] / (df['ret_60d'].abs() + 1)
        print("使用动量因子：20日/60日收益率比值")
    else:
        print(f"未知的动量类型 '{momentum_type}'，使用默认的20日收益率")
        df['factor2_raw'] = df.groupby('股票代码')['收盘价_复权'].pct_change(periods=20) * 100

    # 截断处理，去除极端值（1%~99%分位）
    lower = df['factor2_raw'].quantile(0.01)
    upper = df['factor2_raw'].quantile(0.99)
    df['factor2_raw'] = df['factor2_raw'].clip(lower=lower, upper=upper)
  
    print("因子计算完成")
    return df

# ==================== 过滤条件函数 ====================
def apply_filters(df):
    """应用过滤条件：光底阴线排除、成交额>=500万、市值>=9亿"""
    print("正在应用过滤条件...")
    original_len = len(df)
  
    # 1. 光底阴线（排除）
    if all(col in df.columns for col in ['开盘价', '收盘价', '最低价']):
        df['is_light_bottom_bear'] = (df['开盘价'] > df['收盘价']) & (df['最低价'] == df['收盘价'])
        df = df[~df['is_light_bottom_bear']]
        print(f"过滤光底阴线: {original_len - len(df)} 行被过滤，剩余 {len(df)} 行")
  
    # 2. 日均成交额 >= 500万（20日均值）
    if '成交额' in df.columns:
        df['avg_amount_20d'] = df.groupby('股票代码')['成交额'].rolling(20, min_periods=10).mean().reset_index(level=0, drop=True)
        before = len(df)
        df = df[df['avg_amount_20d'] >= 5_000_000]
        print(f"过滤日均成交额: {before - len(df)} 行被过滤，剩余 {len(df)} 行")
  
    # 3. 市值 >= 9亿
    if '总市值' in df.columns:
        before = len(df)
        df = df[df['总市值'] >= 900_000_000]
        print(f"过滤市值: {before - len(df)} 行被过滤，剩余 {len(df)} 行")
    elif '流通市值' in df.columns:
        before = len(df)
        df = df[df['流通市值'] >= 900_000_000]
        print(f"过滤市值: {before - len(df)} 行被过滤，剩余 {len(df)} 行")
  
    # 移除 NaN（因子计算中产生的）
    df = df.dropna(subset=['factor1_raw', 'factor2_raw'])
    print(f"移除NaN后剩余 {len(df)} 行")
    return df

# ==================== 策略信号生成函数 ====================
def generate_signals(df):
    """综合因子 = 因子1排名（小值优）+ 因子2排名（大值优），等权合成"""
    print("正在生成交易信号...")
    df = df.sort_values(['交易日期', '股票代码'])
    all_dates = sorted(df['交易日期'].unique())
    rebalance_dates = all_dates[::HOLDING_DAYS]   # 每HOLDING_DAYS天调仓
  
    stock_picks = {}
    for i, date in enumerate(rebalance_dates):
        day_data = df[df['交易日期'] == date].copy()
        if len(day_data) < SELECT_NUM:
            continue
  
        # 因子排名（因子1升序，因子2降序）
        day_data['rank1'] = day_data['factor1_raw'].rank(method='min', ascending=True)
        day_data['rank2'] = day_data['factor2_raw'].rank(method='min', ascending=False)
  
        # 转为得分（排名越高越好）
        max_rank = len(day_data)
        day_data['score1'] = max_rank - day_data['rank1'] + 1
        day_data['score2'] = max_rank - day_data['rank2'] + 1
  
        # 综合得分（等权）
        day_data['composite_score'] = 0.5 * day_data['score1'] + 0.5 * day_data['score2']
        day_data['final_rank'] = day_data['composite_score'].rank(method='min', ascending=False)
  
        selected = day_data[day_data['final_rank'] <= SELECT_NUM].copy()
        if len(selected) > 0:
            # 保留股票名称信息
            stock_picks[date] = selected[['股票代码', '股票名称', '收盘价_复权', 'composite_score', 'final_rank', 
                                          'factor1_raw', 'factor2_raw', 'rank1', 'rank2']].to_dict('records')
            if i % 50 == 0:
                print(f"调仓日期 {date}，选中 {len(selected)} 只股票")
  
    print(f"共生成 {len(stock_picks)} 个调仓日期的信号")
    return stock_picks

# ==================== 回测执行函数 ====================
def run_backtest(df, stock_picks):
    print("开始执行回测...")
    cash = INIT_CASH
    current_holdings = {}   # {股票代码: {'buy_price': price, 'shares': shares, 'stock_name': name}}
    holding_days = {}       # {股票代码: 持有天数}
    daily_values = []
    trade_records = []
  
    all_dates = sorted(df['交易日期'].unique())
    rebalance_dates = sorted(stock_picks.keys())
  
    for i, date in enumerate(all_dates):
        # ----- 调仓日：先卖后买 -----
        if date in rebalance_dates:
            # 卖出旧持仓
            if current_holdings:
                day_data = df[df['交易日期'] == date]
                for stock_code, info in current_holdings.items():
                    stock_info = day_data[day_data['股票代码'] == stock_code]
                    if len(stock_info) > 0:
                        sell_price = stock_info.iloc[0]['收盘价_复权']
                        stock_name = stock_info.iloc[0].get('股票名称', stock_code)
                    else:
                        sell_price = info['buy_price']   # 停牌则按买入价
                        stock_name = info.get('stock_name', stock_code)
                    shares = info['shares']
                    sell_amount = sell_price * shares
                    commission = sell_amount * COMMISSION_RATE
                    stamp_tax = sell_amount * STAMP_TAX_RATE
                    cash += sell_amount - commission - stamp_tax
                    trade_records.append({
                        '日期': date, 
                        '股票代码': stock_code, 
                        '股票名称': stock_name,
                        '操作': '卖出',
                        '价格': sell_price, 
                        '数量': shares, 
                        '金额': sell_amount,
                        '持仓天数': holding_days.get(stock_code, 0)
                    })
                current_holdings = {}
                holding_days = {}
  
            # 买入新持仓（等权重）
            picks = stock_picks.get(date, [])
            if picks:
                cash_per_stock = cash / len(picks)
                for pick in picks:
                    stock_code = pick['股票代码']
                    stock_name = pick.get('股票名称', stock_code)
                    buy_price = pick['收盘价_复权']
                    max_shares = int(cash_per_stock / buy_price / (1 + COMMISSION_RATE))
                    if max_shares > 0:
                        buy_amount = max_shares * buy_price
                        commission = buy_amount * COMMISSION_RATE
                        if buy_amount + commission <= cash_per_stock:
                            cash -= (buy_amount + commission)
                            current_holdings[stock_code] = {
                                'buy_price': buy_price, 
                                'shares': max_shares,
                                'stock_name': stock_name
                            }
                            holding_days[stock_code] = 0
                            trade_records.append({
                                '日期': date, 
                                '股票代码': stock_code,
                                '股票名称': stock_name,
                                '操作': '买入',
                                '价格': buy_price, 
                                '数量': max_shares, 
                                '金额': buy_amount,
                                '因子1': pick.get('factor1_raw'), 
                                '因子2': pick.get('factor2_raw'),
                                '综合排名': pick.get('final_rank')
                            })
  
        # 更新持有天数
        for code in list(holding_days.keys()):
            holding_days[code] += 1
  
        # 计算当日总资产
        total_value = cash
        if current_holdings:
            day_data = df[df['交易日期'] == date]
            for stock_code, info in current_holdings.items():
                stock_info = day_data[day_data['股票代码'] == stock_code]
                if len(stock_info) > 0:
                    current_price = stock_info.iloc[0]['收盘价_复权']
                else:
                    current_price = info['buy_price']
                total_value += current_price * info['shares']
  
        daily_values.append({'日期': date, '总资产': total_value, '现金': cash, 
                             '持仓市值': total_value - cash})
        if i % 100 == 0:
            print(f"回测进度: {i+1}/{len(all_dates)}")
  
    # 最后一天强制清仓
    if current_holdings:
        last_date = all_dates[-1]
        day_data = df[df['交易日期'] == last_date]
        for stock_code, info in current_holdings.items():
            stock_info = day_data[day_data['股票代码'] == stock_code]
            if len(stock_info) > 0:
                sell_price = stock_info.iloc[0]['收盘价_复权']
                stock_name = stock_info.iloc[0].get('股票名称', stock_code)
            else:
                sell_price = info['buy_price']
                stock_name = info.get('stock_name', stock_code)
            sell_amount = sell_price * info['shares']
            commission = sell_amount * COMMISSION_RATE
            stamp_tax = sell_amount * STAMP_TAX_RATE
            cash += sell_amount - commission - stamp_tax
            trade_records.append({
                '日期': last_date,
                '股票代码': stock_code,
                '股票名称': stock_name,
                '操作': '卖出',
                '价格': sell_price,
                '数量': info['shares'],
                '金额': sell_amount,
                '持仓天数': holding_days.get(stock_code, 0)
            })
  
    return pd.DataFrame(daily_values), pd.DataFrame(trade_records)

# ==================== 绩效分析函数 ====================
def performance_analysis(daily_values, trade_records, hs300_data=None, all_a_data=None):
    print("\n" + "="*60)
    print("回测绩效分析")
    print("="*60)
    if len(daily_values) == 0:
        print("没有回测数据")
        return None
  
    daily_values = daily_values.sort_values('日期').reset_index(drop=True)
    daily_values['收益率'] = daily_values['总资产'].pct_change()
  
    final_value = daily_values['总资产'].iloc[-1]
    total_return = (final_value / INIT_CASH - 1) * 100
    days = len(daily_values)
    years = days / 252
    annual_return = (final_value / INIT_CASH) ** (1/years) - 1 if years > 0 else 0
    annual_return_pct = annual_return * 100
  
    cum_max = daily_values['总资产'].cummax()
    max_value = cum_max.max()
    max_drawdown = ((daily_values['总资产'] / cum_max - 1) * 100).min()
  
    risk_free = 0.03
    excess = daily_values['收益率'] - risk_free / 252
    sharpe = np.sqrt(252) * excess.mean() / excess.std() if excess.std() > 0 else 0
  
    win_rate = (daily_values['收益率'] > 0).sum() / len(daily_values) * 100
    buy_count = len(trade_records[trade_records['操作'] == '买入']) if len(trade_records) > 0 else 0
    sell_count = len(trade_records[trade_records['操作'] == '卖出']) if len(trade_records) > 0 else 0
  
    print(f"初始资金: {INIT_CASH:,.0f}")
    print(f"最终资金: {final_value:,.2f}")
    print(f"最大净值: {max_value:,.2f}")
    print(f"总收益率: {total_return:.2f}%")
    print(f"年化收益率: {annual_return_pct:.2f}%")
    print(f"最大回撤: {max_drawdown:.2f}%")
    print(f"夏普比率: {sharpe:.4f}")
    print(f"胜率(日): {win_rate:.2f}%")
    print(f"交易天数: {len(daily_values)}")
    print(f"买入次数: {buy_count}")
    print(f"卖出次数: {sell_count}")
    print("="*60)
  
    # 绘制资金曲线
    plot_results(daily_values, hs300_data, all_a_data, 
                 final_value, max_value, annual_return_pct, max_drawdown)
  
    return {
        'final_value': final_value,
        'max_value': max_value,
        'total_return': total_return,
        'annual_return': annual_return_pct,
        'max_drawdown': max_drawdown,
        'sharpe_ratio': sharpe,
        'win_rate': win_rate
    }

这个策略为什么有效？三个层面：

小市值异象：A股最持久的alpha来源，小盘股信息效率低 → 错误定价机会多，机构覆盖少 → 被忽视的标的容易产生超预期爆发
超跌反弹：不追涨，反而选最近20日跌得最惨的——跌透了的小盘股反弹空间最大。这本质是做"均值回归"，而非趋势跟随
高频轮动：3天调仓 + 5只集中，确保始终持有最超跌的小盘股，反弹即换下一批，不死扛，避免坐过山车

两个因子都非常简单、可解释，且逻辑自洽：小市值提供弹性，超跌提供安全边际，高频轮动提供复利加速度。
