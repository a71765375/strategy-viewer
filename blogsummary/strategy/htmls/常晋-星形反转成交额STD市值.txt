一、策略描述

偏好 小市值 + 低成交额波动 + 非反转形态 的股票。这是一个偏向稳健小盘股的策略——剔除那些已经出现底部反转信号的股票（避免追涨），同时要求成交额稳定（排除异常放量/缩量），整体风格偏防御型小市值轮动。

持仓周期：3D

回测时间：2009/01/01 - 2026/06/05

选股数量：10

# ====================================================================================================
# 3️⃣ 策略配置
# ====================================================================================================
strategy = {
    'name': '策略',  # 策略名
    'hold_period': '3D',  # 持仓周期，W 代表周，M 代表月，还支持日频：3D、5D、10D
    'select_num': 10,  # 选股数量，可以是整数，也可以是小数，比如 0.1 表示选取 10% 的股票
    "factor_list": [  # 选股因子列表
        # ** 因子格式说明 **
        # 因子名称（与 '因子库' 文件中的名称一致），排序方式（True 为升序，False 为降序），因子参数，因子权重   
        ('星形反转信号', True, None, 1),
        ('成交额STD', True,5, 1),
        ('总市值对数', True, None, 1),
        # 案例说明：使用'市值.py'因子，从小到大排序（越小越是我想要），None表示无额外参数，后面计算复合选股因子的时候权重为1
        # 可添加多个选股因子
    ],
    "filter_list": [
        ('换手率', 1, 'pct:<=0.8'),
        ('Ret', 5, 'pct:<0.3', True),  # 保留最近5个交易日涨幅最小的前30%股票
    ]  # 过滤因子列表
}

days_listed = 250  # 上市至今交易天数

# excluded_boards = []  # 过滤板块，默认不过滤
excluded_boards = ["kcb", "bj"]  # 同时过滤创业板和科创板和北交所
